using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

[ExecuteAlways]
public class FogLoader : MonoBehaviour
{
    [Header("Fog Material")]
    [Tooltip("Assign your VolumetricFogMat material here")]
    public Material fogMaterial;

    private FullScreenPassRendererFeature _fogFeature;
    private UniversalRendererData _rendererData;
    private bool _initialized = false;

    void OnEnable()
    {
        _initialized = false;
        EnsureFogFeature();
    }

    void OnValidate()
    {
        _initialized = false;
        EnsureFogFeature();
    }

#if UNITY_EDITOR
    void Update()
    {
        // Only run once in editor, not every frame
        if (!Application.isPlaying && !_initialized)
            EnsureFogFeature();
    }
#endif

    void EnsureFogFeature()
    {
        if (fogMaterial == null)
            return;

        _rendererData = GetRendererData();
        if (_rendererData == null)
            return;

        // Find existing valid feature
        _fogFeature = FindFogFeature(_rendererData);

        if (_fogFeature == null)
        {
            _fogFeature = ScriptableObject.CreateInstance<FullScreenPassRendererFeature>();
            _fogFeature.name = "VolumetricFog";
            _rendererData.rendererFeatures.Add(_fogFeature);
            Debug.Log("[VolumetricFog] Created FullScreenPassRendererFeature.");
        }

        _fogFeature.SetActive(true);
        _fogFeature.passMaterial = fogMaterial;
        _rendererData.SetDirty();

        _initialized = true;
    }

    FullScreenPassRendererFeature FindFogFeature(UniversalRendererData data)
    {
        foreach (var feature in data.rendererFeatures)
        {
            if (feature == null)
                continue;
            if (feature is FullScreenPassRendererFeature f && feature.name == "VolumetricFog")
                return f;
        }
        return null;
    }

    UniversalRendererData GetRendererData()
    {
        var pipeline = GraphicsSettings.currentRenderPipeline as UniversalRenderPipelineAsset;
        if (pipeline == null)
            return null;

        var field = typeof(UniversalRenderPipelineAsset).GetField(
            "m_RendererDataList",
            System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance
        );
        if (field == null)
            return null;

        var list = field.GetValue(pipeline) as ScriptableRendererData[];
        if (list == null)
            return null;

        foreach (var data in list)
            if (data is UniversalRendererData urd)
                return urd;

        return null;
    }
}
